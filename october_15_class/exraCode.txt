void demoSocialMedia()
{
	int postLikeCount = 0;
	std::cout << "Current number of likes on a post: " << postLikeCount << std::endl;

	//If ten people already liked the post.
	postLikeCount+= 10; //+=
	std::cout << "After incrementing, like count is " << postLikeCount << std::endl;
}
